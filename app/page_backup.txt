function PatientTable({

    name: string;
    age: number;
    condition: string;
    status: string;

    initials: string;

  setSearch: (value: string) => void;
  onSelect: (name: string) => void;
  onViewAll: () => void;

  return (
    <div className="rounded-2xl border border-slate-100 bg-white p-5 shadow-sm sm:p-6">
      <div className="mb-5 flex flex-col justify-between gap-4 sm:flex-row sm:items-center">

          <h3 className="font-bold">Patient Directory</h3>
          <p className="mt-1 text-xs text-slate-500">Manage and review patient information</p>
        </div>
        <div className="flex flex-wrap gap-2">
          <input
            value={search}
            onChange={(event) => setSearch(event.target.value)}
            placeholder="Search patients..."
            className="min-w-0 rounded-xl border border-slate-200 px-3 py-2.5 text-sm outline-none focus:border-blue-500 sm:w-52"
          />
          <button onClick={onViewAll} className="rounded-xl bg-blue-600 px-4 py-2.5 text-sm font-semibold text-white hover:bg-blue-700">
            View all
          </button>
        </div>
      </div>

        <table className="w-full min-w-[620px] text-left">
          <thead>
            <tr className="border-y border-slate-100 text-xs text-slate-400">

                <td className="py-4 pr-4">

                    <div className="flex h-10 w-10 shrink-0 items-center justify-center rounded-full bg-blue-50 text-xs font-bold text-blue-700">

                    </div>

function MedicationTable() {
  const [completed, setCompleted] = useState<string[]>([]);

  return (
    <div className="overflow-x-auto">
      <table className="w-full min-w-[520px] text-left">
        <thead>
          <tr className="border-y border-slate-100 text-xs text-slate-400">
            <th className="py-4 font-medium">Medication</th>
            <th className="py-4 font-medium">Patient</th>
            <th className="py-4 font-medium">Scheduled</th>
            <th className="py-4 font-medium">Status</th>
          </tr>
        </thead>
        <tbody>
          {medications.map((medication) => {
            const isTaken = completed.includes(medication.name) || medication.status === "Taken";
            return (
              <tr key={medication.name} className="border-b border-slate-50 last:border-0">
                <td className="py-4 pr-4">
                  <p className="text-sm font-semibold">{medication.name}</p>
                  <p className="mt-1 text-xs text-slate-400">{medication.dose}</p>
                </td>
                <td className="py-4 pr-4 text-sm text-slate-600">{medication.patient}</td>
                <td className="py-4 pr-4 text-sm text-slate-500">{medication.time}</td>
                <td className="py-4">
                  <button
                    onClick={() => setCompleted((previous) =>
                      previous.includes(medication.name)
                        ? previous.filter((name) => name !== medication.name)
                        : [...previous, medication.name]
                    )}
                    className={`rounded-full px-3 py-1.5 text-xs font-semibold ${
                      isTaken ? "bg-green-50 text-green-700" : "bg-amber-50 text-amber-700"
                    }`}
                  >
                    {isTaken ? "Taken ✓" : "Mark as taken"}
                  </button>
                </td>
              </tr>
            );
          })}
        </tbody>
      </table>
      <p className="mt-4 text-xs text-slate-400">Medication records are for demonstration only.</p>
    </div>
  );
}export default
